# AAA013.jsx — 화면 진입점 (풀코드)

- 요약/연결 예시 → 퍼블리셔 `AAA013.jsx` 기준 풀코드
- 탭이 없는 화면이므로 `getPaneProps(key)` 대신 `AAA013Tab`에 직접 props 전달

```jsx
import React from 'react';
import AAA013Tab from './tabs/AAA013Tab';
import { useAAA013Search } from './hooks/useAAA013Search';

const SCREEN_ID = 'AAA013';

export default function AAA013() {
    const {
        filters,
        formMethods,
        somCodeList,
        availableSomMonths,
        applyAvailableSomMonths,
        disabledSearchBox,
        initialStockMonth,
        initialSalesResultMonth,
        applyInitialData,
        applySomCodes,
        handleSomMonthChange,
        resetFilters,
        getSearchParams
    } = useAAA013Search();

    return (
        <AAA013Tab
            tabId={SCREEN_ID}
            filters={filters}
            formMethods={formMethods}
            somCodeList={somCodeList}
            availableSomMonths={availableSomMonths}
            applyAvailableSomMonths={applyAvailableSomMonths}
            disabledSearchBox={disabledSearchBox}
            initialStockMonth={initialStockMonth}
            initialSalesResultMonth={initialSalesResultMonth}
            applyInitialData={applyInitialData}
            applySomCodes={applySomCodes}
            handleSomMonthChange={handleSomMonthChange}
            resetFilters={resetFilters}
            getSearchParams={getSearchParams}
        />
    );
}
```
