# AAA013.jsx — 화면 진입점 (풀코드)

메모장3 대비 수정
- 요약/연결 예시 → 퍼블리셔 `AAA013.jsx` 기준 풀코드
- 탭이 없는 화면이므로 `getPaneProps(key)` 대신 `AAA013Tab`에 직접 props 전달

```jsx
import React from 'react';
import AAA013Tab from './tabs/AAA013Tab';
import { useAAA013Search } from './hooks/useAAA013Search';

const SCREEN_ID = 'AAA013';

export default function AAA013() {
    const {
        filters,
        formMethods,
        somCodeList,
        availableSomMonths,
        applyAvailableSomMonths,
        disabledSearchBox,
        initialStockMonth,
        initialSalesResultMonth,
        applyInitialData,
        applySomCodes,
        handleSomMonthChange,
        resetFilters,
        getSearchParams
    } = useAAA013Search();

    return (
        <AAA013Tab
            tabId={SCREEN_ID} // 화면 ID: React Query 캐시 키 구분용 (탭 없는 화면이라 화면 ID 고정)
            filters={filters} // 퍼블리셔 초기 필터 원본: 규약상 유지만 하고 값 변경은 formMethods로만 함
            formMethods={formMethods} // RHF 폼 객체: 검색조건 단일 관리 위해 SearchFilterComp까지 통째로 전달
            somCodeList={somCodeList} // SOM CD Select 옵션 목록: aaa02 조회 결과를 Search 훅이 가공한 값
            availableSomMonths={availableSomMonths} // SOM 월 선택 가능 년월(YYYYMM): aaa05 기준으로 달력 선택 제한
            applyAvailableSomMonths={applyAvailableSomMonths} // aaa05 응답 반영 함수: 조회는 Tab(Fetch 훅), 상태 저장은 Search 훅이라 함수로 넘김
            disabledSearchBox={disabledSearchBox} // 조회 버튼 비활성 여부: SOM CD 없으면 조회 막기 위해 Search 훅에서 관리
            initialStockMonth={initialStockMonth} // 최초 재고조회 월: 이후 월 선택 막는 상한값 (사용자 선택과 무관하게 고정)
            initialSalesResultMonth={initialSalesResultMonth} // 최초 판매실적 월: 재고조회 월과 별도로 적용하는 상한값
            applyInitialData={applyInitialData} // aaa01 응답 반영 함수: 최초 월·주차를 폼에 1회만 세팅
            applySomCodes={applySomCodes} // aaa02 응답 반영 함수: SOM CD 옵션 갱신 + 첫 코드 자동 선택
            handleSomMonthChange={handleSomMonthChange} // SOM 월 변경 시 SOM CD 선택값·옵션 초기화 (재조회 트리거는 Tab이 담당)
            resetFilters={resetFilters} // 초기화 버튼: API 재호출 없이 최초 폼 값·옵션으로 복원
            getSearchParams={getSearchParams} // 조회 시점의 현재 폼 값 읽기: 상태 복사 대신 RHF getValues로 최신값 보장
        />
    );
}
```
