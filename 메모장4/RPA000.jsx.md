# RPA000.jsx — 화면 진입점 (풀코드)

메모장3 대비 수정
- 요약/연결 예시 → 퍼블리셔 `RPA000.jsx` 기준 풀코드
- 탭이 없는 화면이므로 `getPaneProps(key)` 대신 `RPA000Tab`에 직접 props 전달

```jsx
import React from 'react';
import RPA000Tab from './tabs/RPA000Tab';
import { useRPA000Search } from './hooks/useRPA000Search';

const SCREEN_ID = 'RPA000';

export default function RPA000() {
    const {
        filters,
        formMethods,
        somCodeList,
        availableSomMonths,
        applyAvailableSomMonths,
        disabledSearchBox,
        initialStockMonth,
        initialSalesResultMonth,
        applyInitialData,
        applySomCodes,
        handleSomMonthChange,
        resetFilters,
        getSearchParams
    } = useRPA000Search();

    return (
        <RPA000Tab
            tabId={SCREEN_ID}
            filters={filters}
            formMethods={formMethods}
            somCodeList={somCodeList}
            availableSomMonths={availableSomMonths}
            applyAvailableSomMonths={applyAvailableSomMonths}
            disabledSearchBox={disabledSearchBox}
            initialStockMonth={initialStockMonth}
            initialSalesResultMonth={initialSalesResultMonth}
            applyInitialData={applyInitialData}
            applySomCodes={applySomCodes}
            handleSomMonthChange={handleSomMonthChange}
            resetFilters={resetFilters}
            getSearchParams={getSearchParams}
        />
    );
}
```
