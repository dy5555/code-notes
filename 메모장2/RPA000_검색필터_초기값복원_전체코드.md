# RPA000 검색 필터 - 기존 퍼블리싱 구조 유지 버전

> 이 문서는 메모장2에 올렸던 '별도 useRPA000InitialData / RPA000Content' 설계를 대체합니다. 기존 메모장 폴더와 실제 프로젝트 소스는 수정하지 않습니다.

## 목표

- 기존 `RPA000.jsx → RPA000Tab.jsx → SearchFilterComp.jsx` 및 `useRPA000Search.js` 구조 유지
- 최초 월·주차 API 조회 결과를 `initialFiltersRef`에 별도 보관하고 기존 `filters` state도 유지
- 사용자가 폼 필드를 변경해도 `initialFiltersRef.current`는 최초 조회값으로 유지
- 초기화 버튼은 **API 재조회 없이** `setValue()`로 여섯 필드 복원
- SOM 코드 옵션 목록도 최초 조회 목록으로 복원
- `reset()` 미사용
- `useRPA000InitialData`, `RPA000Content` 추가하지 않음

## useRPA000Search.js 전체 예시

```jsx
import { useState, useEffect, useCallback, useRef } from 'react';
import { useForm } from 'react-hook-form';
import dayjs from 'dayjs';
import isoWeek from 'dayjs/plugin/isoWeek';
import isoWeekYear from 'dayjs/plugin/isoWeekYear';

dayjs.extend(isoWeek);
dayjs.extend(isoWeekYear);

const createInitialFilters = () => ({
    stockMonth: null,
    salesDemandWeek: null,
    somMonth: null,
    somCd: null,
    salesResultMonth: null,
    inboundDemandWeek: null
});

export function useRPA000Search() {
    // 기존 퍼블리싱에서 사용하던 filters 유지
    const [filters, setFilters] = useState(createInitialFilters);
    // 초기화 버튼에서 사용할 최초 조회값 별도 보관 (저장 시 렌더링 없음)
    const initialFiltersRef = useRef(null);
    const [somCodeList, setSomCodeList] = useState([]);
    const [initialSomCodeList, setInitialSomCodeList] = useState([]);

    // 기존 useForm 구조 유지
    const formMethods = useForm({
        defaultValues: createInitialFilters()
    });
    const { setValue, getValues } = formMethods;

    // 실제 프로젝트의 기존 API로 교체
    const loadSomCodes = useCallback(async (somMonth) => {
        if (!somMonth) {
            setSomCodeList([]);
            return [];
        }

        const result = await getSomCodeList({ somMonth });
        const options = (result.data ?? []).map((item) => ({
            label: item.plnVer,
            value: item.plnVer
        }));

        setSomCodeList(options);
        return options;
    }, []);

    // SOM 월 변경 시, 선택된 월에 맞는 코드만 새로 조회
    const handleSomMonthChange = useCallback(async (newMonth) => {
        setValue('somMonth', newMonth);
        setValue('somCd', null);
        setSomCodeList([]);

        if (!newMonth) return;

        try {
            const somMonth = dayjs(newMonth).format('YYYYMM');
            const options = await loadSomCodes(somMonth);
            setValue('somCd', options[0]?.value ?? null);
        } catch (error) {
            console.error('SOM 코드 조회 실패', error);
        }
    }, [loadSomCodes, setValue]);

    // 최초 화면 진입 시에만 API 조회
    useEffect(() => {
        let cancelled = false;

        const loadInitialFilters = async () => {
            try {
                // 실제 프로젝트의 기존 월·주차 조회 API로 교체
                const result = await getInitialMonthWeek();
                const initialValues = createInitialFilters();

                (result.data ?? []).forEach((item) => {
                    switch (item.type) {
                        case 'STOCK_YM':
                            initialValues.stockMonth = item.code;
                            break;
                        case 'SALES_DEMAND_YM':
                            initialValues.salesDemandWeek = item.code;
                            break;
                        case 'SOM_YM':
                            initialValues.somMonth = item.code;
                            break;
                        case 'SALES_ACT_YM':
                            initialValues.salesResultMonth = item.code;
                            break;
                        case 'GR_DEMAND_YM':
                            initialValues.inboundDemandWeek = item.code;
                            break;
                        default:
                            break;
                    }
                });

                const options = await loadSomCodes(initialValues.somMonth);
                if (cancelled) return;

                initialValues.somCd = options[0]?.value ?? null;

                // 기존 filters state 유지 + 최초 조회값을 별도 ref에 보관
                setFilters(initialValues);
                initialFiltersRef.current = { ...initialValues };
                setInitialSomCodeList(options);

                // reset() 없이 현재 폼 필드에 조회값 설정
                Object.entries(initialValues).forEach(([key, value]) => {
                    setValue(key, value);
                });
            } catch (error) {
                if (!cancelled) {
                    console.error('초기 검색조건 조회 실패', error);
                }
            }
        };

        loadInitialFilters();
        return () => { cancelled = true; };
    }, [loadSomCodes, setValue]);

    // 초기화 버튼: 저장된 최초 값만 복원, API 호출 없음
    const resetFilters = useCallback(() => {
        const initialValues = initialFiltersRef.current;
        if (!initialValues) return;

        Object.entries(initialValues).forEach(([key, value]) => {
            setValue(key, value);
        });
        setSomCodeList(initialSomCodeList);
    }, [initialSomCodeList, setValue]);

    const getSearchParams = useCallback(() => {
        const params = getValues();

        const formatMonth = (value) => {
            if (!value) return null;
            if (/^\d{6}$/.test(String(value))) return String(value);
            return dayjs(value).format('YYYYMM');
        };

        const formatWeek = (value) => {
            if (!value) return null;
            if (/^\d{6}$/.test(String(value))) return String(value);
            const date = dayjs(value);
            return `${date.isoWeekYear()}${String(date.isoWeek()).padStart(2, '0')}`;
        };

        return {
            ...params,
            stockMonth: formatMonth(params.stockMonth),
            salesDemandWeek: formatWeek(params.salesDemandWeek),
            somMonth: formatMonth(params.somMonth),
            salesResultMonth: formatMonth(params.salesResultMonth),
            inboundDemandWeek: formatWeek(params.inboundDemandWeek)
        };
    }, [getValues]);

    return {
        filters,
        formMethods,
        resetFilters,
        somCodeList,
        handleSomMonthChange,
        getSearchParams
    };
}
```

## 기존 컴포넌트 연결

`RPA000.jsx`에서 기존처럼 `useRPA000Search()`를 **한 번 호출**하고 반환값을 `RPA000Tab.jsx`에 전달합니다. `RPA000Tab.jsx`는 `SearchFilterComp.jsx`에 `formMethods`, `resetFilters`, `somCodeList`, `handleSomMonthChange` 등을 기존 방식으로 전달합니다. 별도 초기 데이터 Hook이나 화면 컴포넌트가 필요 없습니다.

## 꼭 확인할 점

1. `getInitialMonthWeek`, `getSomCodeList`, `item.type`, `item.code`, `item.plnVer`는 **예시 이름**입니다. 실제 API와 응답 필드에 맞춰 변경하세요.
2. `filters`는 기존 화면 호환을 위해 그대로 두고, 초기화 기준값만 `initialFiltersRef.current`에 별도 저장합니다. 입력 중인 현재 값은 `formMethods`에서 관리합니다. `SearchFilterComp`가 `filters`를 직접 표시값으로 쓰고 있다면 실제 연결을 확인해야 합니다.
3. 날짜 선택 컴포넌트가 날짜 객체/밀리초를 요구한다면 API의 `YYYYMM` 값을 `setValue` 전에 해당 형식으로 변환해야 합니다.
4. 위 주차 변환은 ISO 주차 예시이므로 실제 프로젝트의 주차 규칙에 맞춰 조정하세요.
5. 초기 SOM 코드 조회와 사용자의 SOM 월 변경 조회가 겹칠 수 있는 환경에서는 비동기 요청 순서 보호가 추가로 필요할 수 있습니다.
6. 최초 API 조회 완료 전에는 `initialFiltersRef.current`가 `null`이므로 초기화 버튼이 아무 동작도 하지 않습니다.
7. **주의:** `useRef`는 최초값 저장 시 불필요한 렌더링을 막지만, 이미 관찰된 `setValue()` 이후 반복 렌더링을 해결한다고 보장하지는 않습니다. 초기 폼 값 적용 시의 `setValue()`는 그대로 남아 있으므로 반복이 지속되면 폼 필드 바인딩/상위 렌더링 경로를 점검해야 합니다.
