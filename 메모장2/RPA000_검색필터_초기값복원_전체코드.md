# RPA000 검색 필터 - 최초값 복원 설계 (새 버전)

> 기존 `메모장/` 파일은 수정하지 않음. 이 문서는 별도의 새 설계안입니다.

## 요구사항

- 최초 진입 시 월·주차 API 조회, SOM 코드 조회
- 최초 조회값을 React Hook Form의 `defaultValues`로 저장
- 초기화 시 API 재조회 없이 `setValue()`로 6개 필드 복원
- 최초 SOM 코드 옵션 목록도 복원
- `reset()` 사용하지 않음
- 필드: `stockMonth`, `salesDemandWeek`, `somMonth`, `somCd`, `salesResultMonth`, `inboundDemandWeek`

## useRPA000Search.js (예시 전체 코드)

```jsx
import { useState, useEffect, useCallback } from 'react';
import { useForm } from 'react-hook-form';
import dayjs from 'dayjs';

const createInitialFilters = () => ({
    stockMonth: null,
    salesDemandWeek: null,
    somMonth: null,
    somCd: null,
    salesResultMonth: null,
    inboundDemandWeek: null
});

const createFiltersFromResult = (rows) => {
    const values = createInitialFilters();

    rows.forEach((item) => {
        switch (item.type) {
            case 'STOCK_YM':
                values.stockMonth = item.code;
                break;
            case 'SALES_DEMAND_YM':
                values.salesDemandWeek = item.code;
                break;
            case 'SOM_YM':
                values.somMonth = item.code;
                break;
            case 'SALES_ACT_YM':
                values.salesResultMonth = item.code;
                break;
            case 'GR_DEMAND_YM':
                values.inboundDemandWeek = item.code;
                break;
            default:
                break;
        }
    });

    return values;
};

// API를 먼저 조회하고, 조회 완료 후 폼 컴포넌트를 마운트한다.
export function useRPA000InitialData() {
    const [initialData, setInitialData] = useState(null);

    useEffect(() => {
        let cancelled = false;

        const load = async () => {
            try {
                // TODO: 실제 프로젝트 월·주차 API로 교체
                const result = await getInitialMonthWeek();
                const initialValues = createFiltersFromResult(result.data);

                // TODO: 실제 프로젝트 SOM 코드 API로 교체
                const somResult = await getSomCodeList({
                    somMonth: initialValues.somMonth
                });

                const somOptions = (somResult.data ?? []).map((item) => ({
                    label: item.plnVer,
                    value: item.plnVer
                }));

                initialValues.somCd = somOptions[0]?.value ?? null;

                if (cancelled) return;

                setInitialData({
                    filters: initialValues,
                    somCodeList: somOptions
                });
            } catch (error) {
                console.error('초기 검색조건 조회 실패', error);
            }
        };

        load();

        return () => {
            cancelled = true;
        };
    }, []);

    return initialData;
}

export function useRPA000Search(initialData) {
    const formMethods = useForm({
        defaultValues: initialData.filters
    });

    const { setValue, getValues } = formMethods;

    const [somCodeList, setSomCodeList] = useState(
        () => initialData.somCodeList
    );

    const handleSomMonthChange = useCallback(async (newMonth) => {
        setValue('somMonth', newMonth);
        setValue('somCd', null);
        setSomCodeList([]);

        if (!newMonth) return;

        try {
            const somMonth = dayjs(newMonth).format('YYYYMM');

            // TODO: 실제 프로젝트 SOM 코드 API로 교체
            const result = await getSomCodeList({ somMonth });
            const options = (result.data ?? []).map((item) => ({
                label: item.plnVer,
                value: item.plnVer
            }));

            setSomCodeList(options);
            setValue('somCd', options[0]?.value ?? null);
        } catch (error) {
            console.error('SOM 코드 조회 실패', error);
        }
    }, [setValue]);

    const resetFilters = useCallback(() => {
        // 주의: _defaultValues는 RHF 내부 속성으로 공개 API가 아님.
        const initialValues = formMethods.control._defaultValues;

        Object.entries(initialValues).forEach(([key, value]) => {
            setValue(key, value);
        });

        setSomCodeList(initialData.somCodeList);
    }, [formMethods.control, initialData.somCodeList, setValue]);

    const getSearchParams = useCallback(() => getValues(), [getValues]);

    return {
        filters: initialData.filters,
        formMethods,
        resetFilters,
        somCodeList,
        handleSomMonthChange,
        getSearchParams
    };
}
```

## RPA000.jsx 연결 예시

```jsx
function RPA000() {
    const initialData = useRPA000InitialData();

    if (!initialData) {
        return null;
    }

    return <RPA000Content initialData={initialData} />;
}

function RPA000Content({ initialData }) {
    const {
        filters,
        formMethods,
        resetFilters,
        somCodeList,
        handleSomMonthChange,
        getSearchParams
    } = useRPA000Search(initialData);

    return (
        <RPA000Tab
            filters={filters}
            formMethods={formMethods}
            resetFilters={resetFilters}
            somCodeList={somCodeList}
            handleSomMonthChange={handleSomMonthChange}
            getSearchParams={getSearchParams}
        />
    );
}
```

## 적용 전 주의

1. `getInitialMonthWeek`, `getSomCodeList`, `item.type`, `item.code`, `item.plnVer`는 예시이며 실제 API에 맞춰 교체해야 합니다.
2. `formMethods.control._defaultValues`는 React Hook Form의 **비공개 내부 속성**입니다. 라이브러리 버전 변경에 취약하므로 운영 코드에서는 권장하지 않습니다. 더 안전한 대안은 부모의 `initialData.filters`를 그대로 읽어 `setValue`하는 것입니다.
3. 날짜 필드가 문자열 대신 날짜 객체/타임스탬프를 요구한다면 API 응답을 폼 생성 전에 변환해야 합니다.
4. 주차 조회 파라미터 변환은 기존 프로젝트 로직에 맞게 별도 유지해야 합니다.
5. 이 설계는 최초 API 조회 후 폼을 마운트하므로 기존 화면 구조와 달라질 수 있습니다.
